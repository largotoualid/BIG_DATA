 1. تحميل صورة Docker الخاصة بـ Hadoop

docker pull liliasfaxi/hadoop-cluster:latest

 2. إنشاء شبكة Docker خاصة بـ Hadoop

docker network create hadoop

 3. تشغيل حاوية الماستر (hadoop-master)
docker run -itd --net=hadoop -p 9870:9870 -p 8088:8088 --name hadoop-master --hostname hadoop-master liliasfaxi/hadoop-cluster:latest

 4. تشغيل حاويات الـ Workers
docker run -itd -p 8040:8042 --net=hadoop --name hadoop-worker1 --hostname hadoop-worker1 liliasfaxi/hadoop-cluster:latest
docker run -itd -p 8041:8042 --net=hadoop --name hadoop-worker2 --hostname hadoop-worker2 liliasfaxi/hadoop-cluster:latest

 5. الدخول إلى الحاوية master

docker exec -it hadoop-master bash

 6. إنشاء مجلدات داخل HDFS ورفع البيانات

hdfs dfs -mkdir -p /user/root
hdfs dfs -mkdir -p input
hdfs dfs -put purchases.txt input

 7. التأكد من رفع الملف

hdfs dfs -ls input

أضفنت مكتبات Hadoop إلى pom.xml

أنشأت الحزمة hadoop.mapreduce.tp1 وداخلها الملفات:

WordCount.java

TokenizerMapper.java
